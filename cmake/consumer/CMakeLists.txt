
cmake_minimum_required(VERSION 3.20)
project(ServingObservatoryConsumer CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ServingObservatory CONFIG REQUIRED)
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE ServingObservatory::ServingObservatory)
if(WIN32)
  target_link_libraries(consumer PRIVATE ws2_32)
endif()
